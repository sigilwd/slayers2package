local function getChar()
    local char = LocalPlayer.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if root and hum and hum.Health > 0 then
            return char, root, hum
        end
    end
    return nil, nil, nil
end

local function clean(s)

end

local function isQuestNPC(model)
    local nameLower = model.Name:lower()
    if nameLower:find("quest") or nameLower:find("dialogue") or nameLower:find("giver") or nameLower:find("talk") then
        return true
    end
    if model:FindFirstChild("Quest") or model:FindFirstChild("Dialogue") then
        return true
    end
    return false
end

local function isBossSelected(model)
    if isQuestNPC(model) then return false end
    local cm = clean(model.Name)
    for bossName, selected in pairs(Config.SelectedBosses) do
        if selected then
            local cb = cleanBossNames[bossName] or clean(bossName)
            if cm == cb or cm:find(cb, 1, true) then
                return true
            end
        end
    end
    return false
end

local npcCache = {}
local promptCache = {}
local knownBossPositions = {}
local activeWorldChests = {}

local function isIgnoredPrompt(prompt)
    local parentName = (prompt.Parent and prompt.Parent.Name or ""):lower()
    local objectText = prompt.ObjectText:lower()
    local actionText = prompt.ActionText:lower()
    local combined = parentName .. " " .. objectText .. " " .. actionText

    if combined:find("soul") 
    or combined:find("lily") 
    or combined:find("spider") 
    or combined:find("boulder") 
    or combined:find("train") 
    or combined:find("push") 
    or combined:find("heavy") 
    or combined:find("talk") 
    or combined:find("dialogue") then
        return true
    end

    return false
end

local function findRoot(model)
    return model:FindFirstChild("HumanoidRootPart")
        or model.PrimaryPart
        or model:FindFirstChild("Head")
        or model:FindFirstChildWhichIsA("BasePart")
end

local function considerHumanoid(hum)
    local model = hum.Parent
    if model and model:IsA("Model") and not Players:GetPlayerFromCharacter(model) then
        npcCache[hum] = model
        if isBossSelected(model) then
            local r = findRoot(model)
            if r then
                knownBossPositions[clean(model.Name)] = r.Position
            end
        end
    end
end

local function onDescendantAdded(d)
    if d:IsA("Humanoid") then
        considerHumanoid(d)
    elseif d:IsA("ProximityPrompt") then
        if not isIgnoredPrompt(d) then
            promptCache[d] = true
        end
    elseif d:IsA("Model") and (d.Name:find("World Events Chest") or d.Name:find("WorldEventsChest")) then
        activeWorldChests[d] = true
    end
end

local function onDescendantRemoving(d)
    if d:IsA("Humanoid") then
        npcCache[d] = nil
    elseif d:IsA("ProximityPrompt") then
        promptCache[d] = nil
    elseif d:IsA("Model") then
        activeWorldChests[d] = nil
    end
end

connect(Workspace.DescendantAdded, onDescendantAdded)
connect(Workspace.DescendantRemoving, onDescendantRemoving)

end)

local function firePrompt(prompt)
    if fireproximityprompt then
        fireproximityprompt(prompt, 0)
    else
        prompt:InputHoldBegin()
        task.wait(prompt.HoldDuration + 0.01)
        prompt:InputHoldEnd()
    end
end

local function getActiveWorldChest()
    for chest in pairs(activeWorldChests) do
        if chest.Parent then return chest end
    end
    return nil
end

-- Priority World Event Chest Handler
task.spawn(function()
    while scriptRunning do
        task.wait(0.3)
        if Config.AutoChestEvent then
            local chestModel = getActiveWorldChest()
            if chestModel then
                isChestEventActive = true
                clearTarget()
                setStatus("PRIORITY: World Event Chest Found!")

                local chestRoot = findRoot(chestModel)
                if chestRoot then
                    local lastChestPos = chestRoot.Position
                    local _, root = getChar()
                    if root then
                        root.CFrame = chestRoot.CFrame
                        task.wait(0.15)

                        for prompt in pairs(promptCache) do
                            if prompt.Enabled and prompt.Parent and prompt.Parent:IsDescendantOf(chestModel) then
                                firePrompt(prompt)
                                task.wait(0.1)
                            end
                        end

                        local startTime = tick()
                        while tick() - startTime < 8 and chestModel.Parent do
                            local _, currentRoot = getChar()
                            if not currentRoot then break end
                            local itemFound = false
                            for prompt in pairs(promptCache) do
                                if prompt.Enabled and prompt.Parent and prompt.Parent:IsA("BasePart") then
                                    local part = prompt.Parent
                                    if (part.Position - lastChestPos).Magnitude <= 150 then
                                        itemFound = true
                                        currentRoot.CFrame = part.CFrame
                                        task.wait(0.05)
                                        firePrompt(prompt)
                                        task.wait(0.05)
                                    end
                                end
                            end
                            if not itemFound then task.wait(0.2) end
                        end

                        setStatus("Chest despawned: Clearing remaining loot...")
                        local sweepStart = tick()
                        while tick() - sweepStart < 4 do
                            local _, currentRoot = getChar()
                            if not currentRoot then break end
                            local lootFound = false
                            for prompt in pairs(promptCache) do
                                if prompt.Enabled and prompt.Parent and prompt.Parent:IsA("BasePart") then
                                    local part = prompt.Parent
                                    if (part.Position - lastChestPos).Magnitude <= 150 then
                                        lootFound = true
                                        currentRoot.CFrame = part.CFrame
                                        task.wait(0.05)
                                        firePrompt(prompt)
                                        task.wait(0.05)
                                    end
                                end
                            end
                            if not lootFound then break end
                            task.wait(0.1)
                        end
                    end
                end
                isChestEventActive = false
            else
                isChestEventActive = false
            end
        else
            isChestEventActive = false
        end
    end
end)

-- INTEGRATED TARGET & SCOUTING LOOP (Anti-Cheat Friendly / Idle Wait)
task.spawn(function()
    local lastScoutIndex = 1

    while scriptRunning do
        task.wait(0.3)
        if Config.AutoBoss and not isChestEventActive then
            if not targetValid() then
                clearTarget()
                local foundTarget = nil

                -- 1. Check if any selected boss is already loaded in workspace
                for hum, model in pairs(npcCache) do
                    if hum and hum.Parent == model and hum.Health > 0 and isBossSelected(model) then
                        local root = findRoot(model)
                        if root then
                            foundTarget = { model = model, root = root, hum = hum }
                            break
                        end
                    end
                end

                if foundTarget then
                    target.model, target.root, target.hum = foundTarget.model, foundTarget.root, foundTarget.hum
                    knownBossPositions[clean(target.model.Name)] = target.root.Position
                else
                    -- 2. Find selected bosses that have known coordinates
                    local activeBossList = {}
                    for bossName, selected in pairs(Config.SelectedBosses) do
                        if selected and BossCenters[bossName] then
                            table.insert(activeBossList, bossName)
                        end
                    end

                    if #activeBossList > 0 then
                        -- Pick one boss spot to hang out at instead of swapping endlessly
                        if lastScoutIndex > #activeBossList then lastScoutIndex = 1 end
                        local targetBossName = activeBossList[lastScoutIndex]
                        local centerPos = BossCenters[targetBossName]

                        setStatus("Chilling at " .. targetBossName .. " spawn point...")

                        -- Gently hold position in small increments to let anti-cheat settle
                        local char, root = getChar()
                        if char and root then
                            root.CFrame = CFrame.new(centerPos)
                            root.AssemblyLinearVelocity = Vector3.zero
                        end

                        -- Wait 10 seconds at this spot checking if a boss spawns before checking another spot
                        local waitStart = tick()
                        while tick() - waitStart < 10 do
                            if not scriptRunning or not Config.AutoBoss or isChestEventActive then break end
                            
                            -- Check if target loaded while chilling
                            for hum, model in pairs(npcCache) do
                                if hum and hum.Parent == model and hum.Health > 0 and isBossSelected(model) then
                                    local root = findRoot(model)
                                    if root then
                                        foundTarget = { model = model, root = root, hum = hum }
                                        break
                                    end
                                end
                            end

                            if foundTarget then break end
                            task.wait(0.5)
                        end

                        if foundTarget then
                            target.model, target.root, target.hum = foundTarget.model, foundTarget.root, foundTarget.hum
                            knownBossPositions[clean(target.model.Name)] = target.root.Position
                        else
                            -- Rotate to the next boss spawn point only after waiting 10 full seconds
                            lastScoutIndex = lastScoutIndex + 1
                        end
                    end
                end
            end

            if targetValid() then
                local pct = math.floor(target.hum.Health / math.max(target.hum.MaxHealth, 1) * 100)
                setStatus(("Locked: %s (%d%% HP)"):format(target.model.Name, pct))
            end
        elseif not isChestEventActive then
            clearTarget()
            setStatus("Idle")
        end
    end
end)

-- PROXIMITY LOOT LOOP
task.spawn(function()
    while scriptRunning do
        task.wait(0.25)
        local mode = Config.LootMode
        if mode ~= "Off" and not isChestEventActive then
            local _, root = getChar()
            if root then
                local range = Config.PickupRange
                local nearest, nearestPart, nearestDist = nil, nil, math.huge
                for prompt in pairs(promptCache) do
                    if prompt.Enabled and prompt.Parent and prompt.Parent:IsA("BasePart") then
                        local part = prompt.Parent
                        local d = (part.Position - root.Position).Magnitude
                        local reach = (mode == "Pickup (No TP)") and math.min(range, prompt.MaxActivationDistance) or range
                        if d <= reach and d < nearestDist then
                            nearest, nearestPart, nearestDist = prompt, part, d
                        end
                    end
                end

                if nearest then
                    if mode == "Teleport" then
                        isLooting = true
                        root.CFrame = nearestPart.CFrame
                        task.wait(0.05)
                        firePrompt(nearest)
                        task.wait(0.1)
                        isLooting = false
                    else
                        firePrompt(nearest)
                    end
                end
            end
        else
            if not isChestEventActive then isLooting = false end
        end
    end
end)

    for pageName, p in pairs(pages) do
        p.Visible = (pageName == name)
    end
    for btnName, b in pairs(tabButtons) do
        local active = btnName == name
        b.BackgroundColor3 = active and COLORS.accent or COLORS.item
        b.TextColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(180, 185, 200)
    end
end

local function createTabPage(name)
    local page = Instance.new("Frame")
    page.Name = name .. "Page"
    page.Size = UDim2.new(1, 0, 1, 0)
    page.BackgroundTransparency = 1
    page.Visible = false
    page.Parent = PagesFolder
    pages[name] = page

    local btn = button(TabNavBar, name, UDim2.new(0.24, 0, 1, 0), UDim2.new(), COLORS.item)
    btn.TextColor3 = Color3.fromRGB(180, 185, 200)
    tabButtons[name] = btn
    connect(btn.MouseButton1Click, function() selectTab(name) end)
    return page
end

local function createSlider(parent, text, minVal, maxVal, defaultVal, callback, unit, step)
    unit = unit or " studs"
    step = step or 1
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -10, 0, 45)
    frame.BackgroundColor3 = COLORS.card
    frame.Parent = parent
    corner(frame)

    local lbl = label(frame, "", UDim2.new(1, -20, 0, 20), UDim2.new(0, 10, 0, 4))
    local bg = Instance.new("Frame")
    bg.Size = UDim2.new(1, -20, 0, 8)
    bg.Position = UDim2.new(0, 10, 0, 28)
    bg.BackgroundColor3 = Color3.fromRGB(50, 52, 65)
    bg.Parent = frame
    corner(bg, 4)
    local fill = Instance.new("Frame")
    fill.BackgroundColor3 = COLORS.fill
    fill.Parent = bg
    corner(fill, 4)

    local function set(val)
        val = math.clamp(tonumber(val) or minVal, minVal, maxVal)
        val = math.floor(val / step + 0.5) * step
        fill.Size = UDim2.new((val - minVal) / (maxVal - minVal), 0, 1, 0)
        lbl.Text = text .. ": " .. tostring(val) .. unit
        callback(val)
    end
    local function fromX(x)
        local a = math.clamp((x - bg.AbsolutePosition.X) / math.max(bg.AbsoluteSize.X, 1), 0, 1)
        set(minVal + (maxVal - minVal) * a)
    end
    set(defaultVal)

    connect(bg.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            activeSlider = fromX
            fromX(input.Position.X)
        end
    end)
    return set
end

local function createModeSelector(parent, text, options, defaultOpt, callback)

    frame.Parent = parent
    corner(frame)
    label(frame, text, UDim2.new(0.4, 0, 1, 0), UDim2.new(0, 10, 0, 0))

    local current = defaultOpt
    local btn = button(frame, current, UDim2.new(0.55, -10, 0, 24), UDim2.new(0.45, 0, 0.5, -12),
        Color3.fromRGB(60, 65, 85))
    local function set(opt)
        for _, o in ipairs(options) do
            if o == opt then current = opt end
        end
        btn.Text = current
        callback(current)
    end
    set(defaultOpt)
    connect(btn.MouseButton1Click, function()
        local idx = table.find(options, current) or 1
        set(options[(idx % #options) + 1])
    end)
    return set
end

local function isType(v, t) return type(v) == t end

local function applyLoadedConfig(loaded)
    if not isType(loaded, "table") then return end

    local function bool(key, ui)
        if isType(loaded[key], "boolean") and uiElements[ui] then uiElements[ui](loaded[key]) end
    end
    local function num(key, ui)
        if isType(loaded[key], "number") and uiElements[ui] then uiElements[ui](loaded[key]) end
    end

    bool("AutoBoss", "AutoBoss")
    bool("AutoChestEvent", "AutoChestEvent")
    bool("AutoM1", "AutoM1")
    bool("AutoSkills", "AutoSkills")
    bool("AutoBlock", "AutoBlock")
    bool("AntiAFK", "AntiAFK")
    num("YOffset", "YOffset")
    num("SkillDelay", "SkillDelay")
    num("PickupRange", "PickupRange")
    num("TargetFPS", "FpsSlider")

    if loaded.PositionMode == "Above" or loaded.PositionMode == "Underneath" then
        uiElements.PositionMode(loaded.PositionMode)
    end
    if table.find(LOOT_MODES, loaded.LootMode) then
        uiElements.LootMode(loaded.LootMode)
    elseif loaded.AutoLoot == true then
        uiElements.LootMode("Teleport")
    end

    if isType(loaded.ToggleKeybindName, "string") and Enum.KeyCode[loaded.ToggleKeybindName] then
        Config.ToggleKeybind = Enum.KeyCode[loaded.ToggleKeybindName]
        KeybindBtn.Text = Config.ToggleKeybind.Name
    end
    if isType(loaded.SelectedBosses, "table") then
        for _, name in ipairs(MasterBossList) do
            checkboxSetters[name](loaded.SelectedBosses[name] == true)
        end
    end
    if isType(loaded.FPSCapEnabled, "boolean") then
        Config.FPSCapEnabled = loaded.FPSCapEnabled
        applyFps()
    end
end

local refreshConfigList
refreshConfigList = function()
    for _, child in ipairs(ConfigScroll:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end

    local files = {}
    pcall(function()
        if listfiles then files = listfiles(CONFIG_FOLDER) end
    end)

    for _, filePath in ipairs(files) do
        local fileName = filePath:match("([^/\\]+)%.json$")
        if fileName then
            local item = Instance.new("Frame")
            item.Size = UDim2.new(1, -10, 0, 32)
            item.BackgroundColor3 = COLORS.item
            item.Parent = ConfigScroll
            corner(item, 5)
            label(item, fileName, UDim2.new(0.5, 0, 1, 0), UDim2.new(0, 10, 0, 0), nil, 12, COLORS.bright)
            local loadBtn = button(item, "Load", UDim2.new(0.22, 0, 0, 22), UDim2.new(0.53, 0, 0.5, -11), Color3.fromRGB(60, 120, 200), 10)
            local delBtn = button(item, "Delete", UDim2.new(0.22, 0, 0, 22), UDim2.new(0.76, 0, 0.5, -11), Color3.fromRGB(180, 50, 50), 10)

            connect(loadBtn.MouseButton1Click, function()
                local ok, err = pcall(function()
                    applyLoadedConfig(HttpService:JSONDecode(readfile(filePath)))
                end)
                setStatus(ok and ("Loaded config: " .. fileName) or "Config failed to load")
                if not ok then warn("[AutoFarm] " .. tostring(err)) end
            end)
            connect(delBtn.MouseButton1Click, function()
                pcall(function() if delfile then delfile(filePath) end end)
                refreshConfigList()
            end)
        end
    end
end

local function terminateScript()
    scriptRunning = false
    Config.AutoBoss, Config.AutoM1, Config.AutoSkills = false, false, false
    Config.AutoBlock, Config.AntiAFK, Config.LootMode = false, false, "Off"
    clearTarget()
    for _, c in ipairs(connections) do
        pcall(function() c:Disconnect() end)
    end
    table.clear(connections)
    table.clear(npcCache)
    table.clear(promptCache)
    table.clear(activeWorldChests)
    if Config.FPSCapEnabled then setFpsLimit(0) end
    ScreenGui:Destroy()
    print("[AutoFarm] Script terminated and cleaned up.")
end

-- ATTACK LOOP
task.spawn(function()
    while scriptRunning do
        task.wait(0.12)
        if Config.AutoBoss and Config.AutoM1 and targetValid() and not isLooting and not isChestEventActive then
            local char, _, hum = getChar()
            if char then
                local tool = char:FindFirstChildOfClass("Tool")
                if not tool then
                    local backpack = LocalPlayer:FindFirstChild("Backpack")
                    tool = backpack and backpack:FindFirstChildOfClass("Tool")
                    if tool then hum:EquipTool(tool) end
                end
                if tool and tool.Parent == char then tool:Activate() end

                local cam = Workspace.CurrentCamera
                if cam then
                    local vp = cam.ViewportSize
                    local pos = Vector2.new(vp.X / 2, vp.Y / 2)
                    if MainFrame.Visible then
                        local p, s = MainFrame.AbsolutePosition, MainFrame.AbsoluteSize
                        if pos.X > p.X and pos.X < p.X + s.X and pos.Y > p.Y and pos.Y < p.Y + s.Y then
                            pos = Vector2.new(vp.X - 60, vp.Y / 2)
                        end
                    end
                    VirtualInputManager:SendMouseButtonEvent(pos.X, pos.Y, 0, true, game, 0)
                    VirtualInputManager:SendMouseButtonEvent(pos.X, pos.Y, 0, false, game, 0)
                end
            end
        end
    end
end)

-- SKILLS LOOP
task.spawn(function()
    while scriptRunning do
        task.wait(Config.SkillDelay)
        if Config.AutoBoss and Config.AutoSkills and targetValid() and not isLooting and not isChestEventActive then
            for _, keyCode in ipairs(SkillKeys) do
                if not (scriptRunning and Config.AutoBoss and Config.AutoSkills and targetValid() and not isLooting and not isChestEventActive) then break end
                VirtualInputManager:SendKeyEvent(true, keyCode, false, game)
                task.wait(0.03)
                VirtualInputManager:SendKeyEvent(false, keyCode, false, game)
                task.wait(0.08)
            end
        end
    end
end)

-- AUTO BLOCK LOOP
task.spawn(function()
    local blocking = false
    while scriptRunning do
        task.wait(0.15)
        if Config.AutoBoss and Config.AutoBlock and targetValid() and not isLooting and not blocking and not isChestEventActive then
            local attacking = false
            pcall(function()
                local animator = target.hum:FindFirstChildOfClass("Animator")
                if animator then
                    local tracks = animator:GetPlayingAnimationTracks()
                    for i = 1, #tracks do
                        if tracks[i].IsPlaying and tracks[i].Priority == Enum.AnimationPriority.Action then
                            attacking = true
                            break
                        end
                    end
                end
            end)
            if attacking then
                blocking = true
                VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.F, false, game)
                task.wait(0.35)
                VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.F, false, game)
                blocking = false
            end
        end
